\alg{Floyd--Warshall}{All-pairs shortest paths, including negative edges.}{$O(n^3)$ time, $O(n^2)$ memory.}{Guard INF addition; negative diagonal after the run indicates a negative cycle.}{Initialize diagonal to zero and absent edges to INF.}{sections/graph/floyd_warshall.cpp}

\alg{Strongly Connected Components}{Kosaraju SCC + condensation DAG.}{$O(n+m)$; \texttt{condense} $O(m\log m)$.}{Recursive DFS may overflow. Comp ids are topological (edges go low to high id): DP in id order.}{Directed graph; call \texttt{init} per test case.}{sections/graph/scc.cpp}
\alg{LCA Binary Lifting}{LCA, distance, kth ancestor, dfs order \texttt{tin}.}{Build $O(n\log n)$, query $O(\log n)$.}{Call \texttt{build(root)}; recursion depth can overflow on large chain.}{Graph must be a tree; 0-indexed. \texttt{tin} feeds Virtual Tree.}{sections/graph/lca_binary_lifting.cpp}
\alg{Deterministic Tree Canonization}{Rooted/unrooted tree-isomorphism ids without hash collisions.}{$O(n\log n)$.}{Compare trees with the same \texttt{TreeCanon} dictionary; recursion depth.}{AHU child-id sorting; handles one or two tree centers.}{sections/graph/tree_hashing.cpp}
